cmake_minimum_required(VERSION 3.31)
project(mlc_llvm_targets LANGUAGES C CXX)
set(ZLIB_USE_STATIC_LIBS ON)
find_package(LLVM CONFIG REQUIRED)
# Exercise the executable discovery used by the FFI addon, including pip's
# console launcher on Windows where the SDK contains only a Python script.
find_program(LLVM_CONFIG_EXE NAMES llvm-config HINTS "${LLVM_TOOLS_BINARY_DIR}" REQUIRED)
execute_process(COMMAND "${LLVM_CONFIG_EXE}" --prefix
  OUTPUT_VARIABLE reported_prefix OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
file(REAL_PATH "${reported_prefix}" reported_prefix)
file(REAL_PATH "${LLVM_INSTALL_PREFIX}" expected_prefix)
if(NOT reported_prefix STREQUAL expected_prefix)
  message(FATAL_ERROR "llvm-config resolved outside the selected SDK: ${reported_prefix}")
endif()
llvm_map_components_to_libnames(libraries all)
add_executable(mlc_llvm_targets targets.cpp)
target_compile_features(mlc_llvm_targets PRIVATE cxx_std_17)
target_include_directories(mlc_llvm_targets SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(definitions NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(mlc_llvm_targets PRIVATE ${definitions})
if(MLC_LLVM_STATIC_PROFILE STREQUAL "all")
  target_compile_definitions(mlc_llvm_targets PRIVATE MLC_LLVM_EXPECT_ALL_PROFILE=1)
else()
  target_compile_definitions(mlc_llvm_targets PRIVATE MLC_LLVM_EXPECT_NATIVE_${LLVM_NATIVE_ARCH}=1)
endif()
target_link_libraries(mlc_llvm_targets PRIVATE ${libraries})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(mlc_llvm_targets PRIVATE "LINKER:-z,defs")
endif()
